def classify(features):
    proba = lr.predict_proba([v])[0][1]

    if proba >= 0.5:
        source = "lr"
    elif rules.matches(features):
        source = "rules"
    else:
        source = "rules-fallback"

    return ClassifyResponseV1(
        label="ransomware" if proba>=0.5 else "benign",
        confidence=float(proba),
        source=source,
    )
